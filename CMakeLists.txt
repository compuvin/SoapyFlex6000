cmake_minimum_required(VERSION 3.16)
project(SoapyFlex6000 LANGUAGES CXX)

find_package(SoapySDR CONFIG REQUIRED)
include(GNUInstallDirs)
add_library(SoapyFlex6000 MODULE src/Flex6000.cpp)
target_compile_features(SoapyFlex6000 PRIVATE cxx_std_17)
target_compile_options(SoapyFlex6000 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(SoapyFlex6000 PRIVATE SoapySDR)
set_target_properties(SoapyFlex6000 PROPERTIES PREFIX "")
install(TARGETS SoapyFlex6000 DESTINATION "${CMAKE_INSTALL_LIBDIR}/SoapySDR/modules${SOAPY_SDR_ABI_VERSION}")

option(FLEX6000_BUILD_OFFLINE_TESTS "Build offline DSP and protocol checks" OFF)
if(FLEX6000_BUILD_OFFLINE_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(flex6000_iq_to_audio tests/iq_to_audio.cpp)
    target_compile_features(flex6000_iq_to_audio PRIVATE cxx_std_17)
    target_compile_options(flex6000_iq_to_audio PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME flex6000_iq_to_audio COMMAND flex6000_iq_to_audio)
    add_executable(flex6000_mock_radio_tx tests/mock_radio_tx.cpp)
    target_compile_features(flex6000_mock_radio_tx PRIVATE cxx_std_17)
    target_compile_options(flex6000_mock_radio_tx PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(flex6000_mock_radio_tx PRIVATE SoapySDR Threads::Threads)
    add_test(NAME flex6000_mock_radio_tx COMMAND flex6000_mock_radio_tx)
    set_tests_properties(flex6000_mock_radio_tx PROPERTIES
        ENVIRONMENT "SOAPY_SDR_PLUGIN_PATH=${CMAKE_CURRENT_BINARY_DIR}"
        TIMEOUT 15)
endif()
